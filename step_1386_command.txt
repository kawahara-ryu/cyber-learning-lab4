function escapeHtml(str) {
  return str.replace(/[&<>"']/g, function(m) {
    return {
      '&': '&amp;',
      '<': '&lt;',
      '>': '&gt;',
      '"': '&quot;',
      "'": '&#39;'
    }[m];
  });
}

function highlightSyntax(code) {
  const strPlaceholders = [];
  let res = code.replace(/('(?:\\\\'|[^'])*'|"(?:\\\\"|[^"])*")/g, (m) => {
    strPlaceholders.push(m);
    return \`___STR_\${strPlaceholders.length - 1}___\`;
  });

  res = res.replace(/\\b(for|while|in|if|else|break)\\b/g, (match) => {
    if (['for', 'while', 'break', 'if', 'else'].includes(match)) {
      return \`<span class="hl-keyword keyword-help" data-word="\${match}">\${match}</span>\`;
    }
    return \`<span class="hl-keyword">\${match}</span>\`;
  });
  res = res.replace(/\\b(range|print|int|str|len|input)\\b/g, (match) => {
    if (['range', 'print', 'len', 'input', 'int'].includes(match)) {
      return \`<span class="hl-func keyword-help" data-word="\${match}">\${match}</span>\`;
    }
    return \`<span class="hl-func">\${match}</span>\`;
  });
  
  res = res.replace(/\\.(append|extend)\\b/g, (match, p1) => {
    return \`.\<span class="hl-func keyword-help" data-word="\${p1}">\${p1}</span>\`;
  });

  res = res.replace(/\\b(\\d+)\\b/g, '<span class="hl-number">$1</span>');

  res = res.replace(/___STR_(\\d+)___/g, (match, p1) => {
    return \`<span class="hl-string">\${strPlaceholders[p1]}</span>\`;
  });
  return res;
}

function formatCode(code) {
  return highlightSyntax(escapeHtml(code));
}

  });
}

function playStep() {

    return;

      createListBox(val, i, 400); // Animation start Y
    });
  } else if (action.type === 'append') {
    const curCount = document.querySelectorAll('.list-box').length;
    createListBox(action.value, curCount, 400);
  } else if (action.type === 'extend') {
    const curCount = document.querySelectorAll('.list-box').length;
    action.value.forEach((val, i) => {
      createListBox(val, curCount + i, 400);
    });
  } else if (action.type === 'replace') {
    const boxes = document.querySelectorAll('.list-box');
    if (boxes[action.index]) {
      // Create new box above it and drop
      createListBox(action.value, action.index, 400, true);
    }
  } else if (action.type === 'print') {

function addTraceRow(round, lineNo, code, varVal, cond, output) {
  const tbody = document.getElementById('trace-tbody');
  if (!tbody) return;
  const tr = document.createElement('tr');
  
  if (Array.isArray(round)) {
    const highlightClass = lineNo || '';
    if (highlightClass) tr.className = 'new-row ' + highlightClass;
    else tr.className = 'new-row';
    
    round.forEach((val, idx) => {
      const td = document.createElement('td');
      if (idx === 0) {
        td.style.color = 'var(--neon-cyan)';
        td.style.fontWeight = 'bold';
      } else if (idx === round.length - 1) {
        td.style.color = 'var(--neon-green)';
      }
      td.style.fontFamily = "'Fira Code', monospace";
      td.innerHTML = (val !== undefined && val !== null) ? val : '-';
      tr.appendChild(td);
    });
  } else {
    tr.className = 'new-row';
    tr.innerHTML = \`
      <td>\${round !== undefined ? round : '-'}</td>
      <td>\${code || '-'}</td>
      <td>\${varVal || '-'}</td>
    \`;
  }
  
  tbody.appendChild(tr);
  const tableContainer = tbody.parentElement ? tbody.parentElement.parentElement : null;
  if (tableContainer) {
    tableContainer.scrollTop = tableContainer.scrollHeight;
  }
}
